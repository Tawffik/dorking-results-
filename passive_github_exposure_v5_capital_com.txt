# ---------------- HELPERS ----------------
def utc_now():
    return datetime.now(timezone.utc).isoformat()

def safe_slug(value):
    value = re.sub(r"[^A-Za-z0-9.-]+", "-", str(value))
    return value.strip("._-") or "target"

def canonical_domain(value):
    value = str(value or "").strip().lower()
    value = re.sub(r"^https?://", "", value)
    value = value.split("/")[0].split(":")[0]
    return value.strip(".")

def write_json(path, data):
    with open(path, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2, ensure_ascii=False)

def write_csv(path, rows, fieldnames):
    with open(path, "w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(
            f, fieldnames=fieldnames, extrasaction="ignore"
        )
        writer.writeheader()
        for row in rows:
            out = dict(row)
            for key, value in list(out.items()):
                if isinstance(value, (list, dict)):
                    out[key] = json.dumps(value, ensure_ascii=False)
            writer.writerow(out)

def result_file_url(item):
    url = item.get("html_url")
    if url:
        return url
    repo = (item.get("repository") or {}).get("full_name", "")
    path = item.get("path", "")
    if repo and path:
        return f"https://github.com/{repo}/blob/HEAD/{path}"
    return ""

def is_noise_path(path: str) -> bool:
    p = (path or "").lower()
    if p.endswith((".md", ".rst", ".txt", ".html")):
        # keep only if path looks secret-related
        if not any(x in p for x in ("secret", "credential", "password", ".env")):
            return True
    return any(m in p for m in EXCLUDE_PATH_MARKERS)

def is_high_value_path(path: str) -> bool:
    p = (path or "").lower()
    return any(m in p for m in HIGH_VALUE_PATH_MARKERS)

def likely_example_path(path):
    return is_noise_path(path)

def build_queries(target):
    domain = canonical_domain(target["domain"])
    aliases = []
    for value in [domain] + (target.get("aliases") or []):
        value = canonical_domain(value)
        if value and value not in aliases:
            aliases.append(value)

    orgs = []
    for value in target.get("github_orgs") or []:
        value = str(value).strip()
        if value and value not in orgs:
            orgs.append(value)

    queries = []
    seen = set()

    def add(category, query, basis):
        query = " ".join(str(query).split()).strip()
        if not query or len(query) > MAX_QUERY_LENGTH:
            return
        key = query.lower()
        if key in seen:
            return
        seen.add(key)
        queries.append({
            "category": category,
            "query": query,
            "evidence_basis": basis,
        })

    for d in aliases:
        # Domain context (light)
        add("domain_reference", f'"{d}"', "exact domain reference")
        add("domain_url_reference", f'"https://{d}"', "URL reference")

        # High-signal filenames
        for filename in HIGH_SIGNAL_FILENAME:
            add(
                "filename_indicator",
                f'"{d}" {filename}',
                "domain + sensitive filename",
            )

        # High-signal credentials
        for pattern in HIGH_SIGNAL_CREDENTIAL:
            add(
                "credential_indicator",
                f'"{d}" {pattern}',
                "domain + credential indicator",
            )

        # Cloud / CI
        for pattern in CLOUD_CI_PATTERNS:
            add(
                "cloud_ci_indicator",
                f'"{d}" {pattern}',
                "domain + cloud/CI indicator",
            )

        # Capital / trading oriented extras
        for pattern in (
            '"X-CAP-API-KEY"',
            '"CAPITAL_API_KEY"',
            '"apiKey"',
            '"CST"',
            '"securityToken"',
        ):
            add(
                "credential_indicator",
                f'"{d}" {pattern}',
                "domain + capital/trading credential indicator",
            )

        # Staging / production context with env files
        add(
            "domain_staging_context",
            f'"{d}" "staging" filename:.env',
            "domain + staging env",
        )
        add(
            "domain_staging_context",
            f'"{d}" "production" filename:.env',
            "domain + production env",
        )

    # Organization-scoped (strongest when org is real)
    for org in orgs:
        add("org_reference", f"org:{org}", "verified organization scope")
        for filename in HIGH_SIGNAL_FILENAME:
            add(
                "org_filename_indicator",
                f"org:{org} {filename}",
                "verified organization + sensitive filename",
            )
        for pattern in HIGH_SIGNAL_CREDENTIAL[:12]:
            add(
                "org_credential_indicator",
                f"org:{org} {pattern}",
                "verified organization + credential indicator",
            )
        for d in aliases:
            add(
                "org_domain_reference",
                f'org:{org} "{d}"',
                "verified organization + target domain",
            )

    return queries

# ---------------- GITHUB API ----------------
def github_search_code(query, page):
    url = f"{GITHUB_API}/search/code"
    params = {"q": query, "per_page": PER_PAGE, "page": page}
    last_error = None
    for attempt in range(MAX_RETRIES):
        try:
            response = SESSION.get(
                url, params=params, timeout=REQUEST_TIMEOUT
            )
            if response.status_code == 200:
                try:
                    return response.json(), None, response.headers
                except ValueError:
                    return (
                        None,
                        "HTTP 200 but response was not valid JSON",
                        response.headers,
                    )
            body = (response.text or "")[:250]
            last_error = f"HTTP {response.status_code}: {body}"
            rate_limited = (
                response.status_code == 403
                and (
                    response.headers.get("X-RateLimit-Remaining") == "0"
                    or "rate limit" in body.lower()
                    or "secondary rate" in body.lower()
                )
            )
            retryable = response.status_code in (
                403, 429, 500, 502, 503, 504
            )
            if attempt < MAX_RETRIES - 1 and (retryable or rate_limited):
                delay = min(90, 2 ** (attempt + 1))
                retry_after = response.headers.get("Retry-After", "")
                reset = response.headers.get("X-RateLimit-Reset", "")
                if retry_after.isdigit():
                    delay = max(delay, min(90, int(retry_after)))
                elif rate_limited and reset.isdigit():
                    remaining = max(1, int(reset) - int(time.time()))
                    delay = max(delay, min(90, remaining))
                time.sleep(delay)
                continue
            return None, last_error, response.headers
        except requests.RequestException as exc:
            last_error = f"{type(exc).__name__}: {str(exc)[:200]}"
            if attempt < MAX_RETRIES - 1:
                time.sleep(min(30, 2 ** (attempt + 1)))
    return None, last_error or "Unknown request error", {}

# ---------------- TRIAGE (IMPROVED) ----------------
def triage_result(path, categories, queries, target_domains):
    p = (path or "").lower()
    q = " ".join(queries or []).lower()
    categories = set(categories or [])
    score = 0
    reasons = []

    if is_high_value_path(p):
        score += 25
        reasons.append("high-value config path")

    if any(domain and domain in q for domain in target_domains):
        score += 15
        reasons.append("query includes target domain")

    if categories.intersection({
        "credential_indicator",
        "org_credential_indicator",
        "cloud_ci_indicator",
    }):
        score += 15
        reasons.append("credential/cloud configuration search hit")

    if categories.intersection({
        "filename_indicator",
        "org_filename_indicator",
    }):
        score += 10
        reasons.append("sensitive filename search hit")

    if is_noise_path(p):
        score -= 30
        reasons.append("noise/example/test path")

    if p.endswith((".md", ".rst", ".txt", ".html")):
        score -= 8
        reasons.append("documentation-like extension")

    score = max(0, min(100, score))

    if score >= 40:
        tier = "A_REVIEW_FIRST"
    elif score >= 25:
        tier = "B_REVIEW"
    elif score >= 12:
        tier = "C_CONTEXT"
    else:
        tier = "D_LOW_SIGNAL"

    return score, tier, reasons

all_findings = {}
all_query_logs = []
target_summaries = []

for target in TARGETS:
    if not target.get("authorized", False):
        print("Skipped target: authorized must be True.")
        continue

    domain = canonical_domain(target.get("domain"))
    label = str(target.get("label") or domain).strip()
    if not domain:
        print(f"Skipped target with empty domain: {label}")
        continue

    target_dir = os.path.join(RUN_DIR, safe_slug(label))
    os.makedirs(target_dir, exist_ok=True)

    print("\n" + "=" * 65)
    print(f"Target: {label} | {domain}")
    print("=" * 65)

    queries = build_queries(target)
    findings_map = {}
    query_logs = []

    for qindex, qitem in enumerate(queries, start=1):
        query = qitem["query"]
        category = qitem["category"]
        print(f"[{qindex}/{len(queries)}] {category}: {query}")

        log = {
            "target_domain": domain,
            "target_label": label,
            "category": category,
            "query": query,
            "evidence_basis": qitem["evidence_basis"],
            "started_at_utc": utc_now(),
            "pages_attempted": 0,
            "items_returned": 0,
            "reported_total_count": None,
            "incomplete_results": None,
            "errors": [],
            "page_cap_reached": False,
            "coverage_status": "NOT_COMPLETED",
            "rate_limit_remaining": None,
        }

        for page in range(1, MAX_PAGES_PER_QUERY + 1):
            log["pages_attempted"] = page
            payload, error, headers = github_search_code(query, page)
            if headers:
                log["rate_limit_remaining"] = headers.get(
                    "X-RateLimit-Remaining"
                )
            if error:
                log["errors"].append({"page": page, "error": error})
                print(f"  API issue on page {page}: {error}")
                break

            items = payload.get("items") or []
            total = payload.get("total_count")
            log["reported_total_count"] = total
            log["incomplete_results"] = payload.get("incomplete_results")
            log["items_returned"] += len(items)

            for item in items:
                repo_data = item.get("repository") or {}
                repo = repo_data.get("full_name") or ""
                path = item.get("path") or ""
                url = result_file_url(item)
                if not repo or not path or not url:
                    continue

                # Drop obvious noise early
                if is_noise_path(path):
                    continue

                identity = (repo.lower(), path.lower())
                if identity not in findings_map:
                    findings_map[identity] = {
                        "target_domain": domain,
                        "target_label": label,
                        "repository": repo,
                        "path": path,
                        "file_url": url,
                        "matched_categories": [],
                        "matched_queries": [],
                        "evidence_bases": [],
                        "search_result_count": 0,
                        "review_status": "UNREVIEWED",
                        "confirmed_vulnerability": False,
                        "secret_value_collected": False,
                        "metadata_only": True,
                    }
                row = findings_map[identity]
                row["search_result_count"] += 1
                if category not in row["matched_categories"]:
                    row["matched_categories"].append(category)
                if query not in row["matched_queries"]:
                    row["matched_queries"].append(query)
                if qitem["evidence_basis"] not in row["evidence_bases"]:
                    row["evidence_bases"].append(qitem["evidence_basis"])

            if len(items) < PER_PAGE:
                break
            if page == MAX_PAGES_PER_QUERY:
                log["page_cap_reached"] = True

        total = log["reported_total_count"]
        if log["errors"]:
            log["coverage_status"] = (
                "PARTIAL_RESULTS_WITH_ERRORS"
                if log["items_returned"]
                else "FAILED"
            )
        elif log["incomplete_results"] is True:
            log["coverage_status"] = "INCOMPLETE_RESULTS_FLAGGED"
        elif log["page_cap_reached"] or (
            total is not None
            and total > log["items_returned"]
            and log["pages_attempted"] >= MAX_PAGES_PER_QUERY
        ):
            log["page_cap_reached"] = True
            log["coverage_status"] = "RESULT_WINDOW_CAPPED"
        else:
            log["coverage_status"] = (
                "SEARCH_COMPLETED_NOT_PROOF_OF_EXHAUSTIVENESS"
            )

        log["finished_at_utc"] = utc_now()
        query_logs.append(log)
        all_query_logs.append(log)
        time.sleep(PAUSE_BETWEEN_QUERIES)

    domains_for_score = [
        canonical_domain(x)
        for x in [domain] + (target.get("aliases") or [])
    ]

    findings = list(findings_map.values())
    for row in findings:
        score, tier, reasons = triage_result(
            row["path"],
            row["matched_categories"],
            row["matched_queries"],
            domains_for_score,
        )
        row["triage_score"] = score
        row["review_queue"] = tier
        row["triage_reasons"] = reasons

    findings.sort(key=lambda row: (
        -row["triage_score"],
        row["repository"].lower(),
        row["path"].lower(),
    ))

    a_leads = [r for r in findings if r["review_queue"] == "A_REVIEW_FIRST"]
    print(f"Unique leads: {len(findings)} | A_REVIEW_FIRST: {len(a_leads)}")

    fields = [
        "target_domain", "target_label", "repository", "path", "file_url",
        "matched_categories", "matched_queries", "evidence_bases",
        "search_result_count", "triage_score", "review_queue",
        "triage_reasons", "review_status", "confirmed_vulnerability",
        "secret_value_collected", "metadata_only",
    ]

    write_json(os.path.join(target_dir, "findings.json"), findings)
    write_json(os.path.join(target_dir, "query_log.json"), query_logs)
    write_csv(os.path.join(target_dir, "findings.csv"), findings, fields)
    write_json(os.path.join(target_dir, "a_review_first.json"), a_leads)

    with open(
        os.path.join(target_dir, "review_queue.txt"),
        "w",
        encoding="utf-8",
    ) as f:
        f.write(
            "PASSIVE GITHUB REVIEW QUEUE — UNCONFIRMED LEADS\n"
            "Metadata only. No secret values collected or tested.\n"
            "Triage score is not severity or vulnerability confirmation.\n\n"
        )
        for row in findings:
            f.write(
                f"[{row['review_queue']} | triage={row['triage_score']}] "
                f"{row['repository']} | {row['path']}\n"
                f"URL: {row['file_url']}\n"
                f"Categories: {json.dumps(row['matched_categories'], ensure_ascii=False)}\n"
                f"Matched queries: {len(row['matched_queries'])}\n"
                f"Review status: {row['review_status']}\n\n"
            )

    with open(
        os.path.join(target_dir, "a_review_first.txt"),
        "w",
        encoding="utf-8",
    ) as f:
        f.write("A_REVIEW_FIRST ONLY — START HERE\n\n")
        for row in a_leads:
            f.write(
                f"[{row['triage_score']}] {row['repository']} | {row['path']}\n"
                f"{row['file_url']}\n\n"
            )

    with open(
        os.path.join(target_dir, "report.md"),
        "w",
        encoding="utf-8",
    ) as f:
        f.write(f"# Passive GitHub Exposure Review — {label}\n\n")
        f.write(f"- Target domain: `{domain}`\n")
        f.write(f"- Run UTC: `{RUN_STAMP}`\n")
        f.write(f"- Queries generated: {len(queries)}\n")
        f.write(f"- Queries logged: {len(query_logs)}\n")
        f.write(f"- Unique repository/path leads: {len(findings)}\n")
        f.write(f"- A_REVIEW_FIRST: {len(a_leads)}\n")
        f.write(
            f"- Queries with errors: "
            f"{sum(bool(x['errors']) for x in query_logs)}\n"
        )
        f.write(
            f"- Capped queries: "
            f"{sum(x['page_cap_reached'] for x in query_logs)}\n\n"
        )
        f.write(
            "## Coverage caveat\n\n"
            "A completed query does not prove exhaustive coverage. "
            "Review query_log.json before interpreting zero results.\n\n"
        )
        f.write("## A_REVIEW_FIRST leads\n\n")
        for row in a_leads:
            f.write(
                f"- **({row['triage_score']})** "
                f"[{row['repository']} — {row['path']}]({row['file_url']})\n"
            )

    summary = {
        "target_domain": domain,
        "target_label": label,
        "queries_generated": len(queries),
        "queries_run": len(query_logs),
        "unique_repository_path_leads": len(findings),
        "a_review_first_count": len(a_leads),
        "queries_with_errors": sum(bool(x["errors"]) for x in query_logs),
        "queries_with_incomplete_results": sum(
            x["incomplete_results"] is True for x in query_logs
        ),
        "queries_capped": sum(bool(x["page_cap_reached"]) for x in query_logs),
        "coverage_is_exhaustive": False,
    }
    target_summaries.append(summary)

    for row in findings:
        key = (
            row["target_domain"],
            row["repository"].lower(),
            row["path"].lower(),
        )
        if key not in all_findings:
            all_findings[key] = row
        else:
            existing = all_findings[key]
            for field in (
                "matched_categories",
                "matched_queries",
                "evidence_bases",
            ):
                for value in row[field]:
                    if value not in existing[field]:
                        existing[field].append(value)
            existing["search_result_count"] += row["search_result_count"]
            if row["triage_score"] > existing["triage_score"]:
                existing["triage_score"] = row["triage_score"]
                existing["review_queue"] = row["review_queue"]
                existing["triage_reasons"] = row["triage_reasons"]

    print(f"Saved: {target_dir}")
    time.sleep(PAUSE_BETWEEN_TARGETS)
